{{- /* Language switcher: one compact link per configured language (the language code, e.g. EN / UK). Links to the current page's translation when it exists. Otherwise it links to the nearest ancestor section that IS translated, so the reader stays close to where they were, and the title says the page itself is not translated. The language's home is the last resort. Iterating hugo.Sites makes each language's home resolve correctly even with the default language served at the root. Renders nothing when only one language is configured. */ -}} {{- if gt (len hugo.Sites) 1 -}}
{{- range hugo.Sites -}} {{- $s := . -}} {{- $lang := $s.Language.Lang -}} {{- $href := "" -}} {{- range $.AllTranslations -}} {{- if eq .Language.Lang $lang }}{{ $href = .RelPermalink }}{{ end -}} {{- end -}} {{- $missing := not $href -}} {{- if $missing -}} {{- range $.Ancestors -}} {{- if not $href -}} {{- range .AllTranslations -}} {{- if eq .Language.Lang $lang }}{{ $href = .RelPermalink }}{{ end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- $href = $href | default $s.Home.RelPermalink -}} {{- $cur := eq $lang site.Language.Lang -}} {{- $title := $s.Language.Label -}} {{- if $missing }}{{ $title = printf "%s - %s" $title (i18n "translation_missing") }}{{ end -}} {{ $lang | upper }} {{- end -}}
{{- end -}}
